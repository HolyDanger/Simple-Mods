cmake_minimum_required(VERSION 3.20)
project(SimpleMods VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(Probe src/probe.cpp)
add_executable(UiInspect src/ui_inspect.cpp)
target_link_libraries(Probe PRIVATE psapi)
add_subdirectory(vendor/minhook)
set(IMGUI vendor/imgui)
add_library(neighbor_imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_include_directories(neighbor_imgui PUBLIC ${IMGUI})
target_link_libraries(neighbor_imgui PUBLIC d3d11 d3dcompiler dwmapi)
add_library(NeighborLoader SHARED src/proxy.cpp src/runtime.cpp src/bridge.cpp src/xinput.def)
target_include_directories(NeighborLoader PRIVATE include src)
target_link_libraries(NeighborLoader PRIVATE neighbor_imgui minhook d3d11 dxgi user32)
set_target_properties(NeighborLoader PROPERTIES OUTPUT_NAME xinput1_3)
add_executable(NeighborAttach src/attach.cpp)
target_link_libraries(NeighborAttach PRIVATE psapi)
add_executable(NeighborTestHost WIN32 src/testhost.cpp)
target_link_libraries(NeighborTestHost PRIVATE d3d11 dxgi)
add_library(ExampleMod SHARED SDK/ExampleMod.cpp)
target_include_directories(ExampleMod PRIVATE include)
add_executable(ModContractTest src/contract_test.cpp)
target_include_directories(ModContractTest PRIVATE include)
